### 1. Count file lines 

wc -l data/fasta/407228326.fasta data/fasta/407228412.fasta data/fasta/e_coli.fasta

# note: wc = word count; -l = lines format. Counts newline characters in the 
# three input files. 
# output: 
#     27 data/fasta/407228326.fasta
#    127 data/fasta/407228412.fasta
#  78104 data/fasta/e_coli.fasta
#  78258 total
# note: 407228326.fasta and 407228412.fasta are not newline-terminated (not ending with \n) 
# which means that the last line isn't counted as one. 
# Thus, although there are x lines, wc -l has an output of x-1.


###2 Display the E. coli sequence lines
tail -n +2 data/fasta/e_coli.fasta 
# note: print the n last lines of the file (tail -n) starting from the second line onwards (+2).
# note: header is excluded because it is the species name (don't want to include that in sequence).
# Header was line 1 and was thus excluded by the +2 part of the command.


###3 Count sequence length 
grep -v ">" data/fasta/e_coli.fasta | tr -d "\n" | wc -c

#note: grep = read input file (../data/fasta/e_coli.fasta)
#note: -v "^>" = exclude lines containing >
#note: tr -d "\n" = remove newline characters
#note: wc -c = count all characters
#note: sequence was 4686136 characters long 

 
###4 Count a literal motif 

grep -v ">" data/fasta/e_coli.fasta | tr -d "\n" | grep -o "ATGC" | wc -l
#note: same concept as previous command except for:
#note: grep -o "ATGC" = finds each "ATGC" motif 
#note: wc -l = counts amount matches that grep -o "ATGC" found 
#note: 21968 motifs found 

###5 Calculate the AT/GC ratio 
a=$(grep -v "^>" data/fasta/e_coli.fasta | tr -d "\n" | grep -o "A" | wc -l)
#note: same concept as previous step. added a=$() to store the output of the command as a 

t=$(grep -v "^>" data/fasta/e_coli.fasta | tr -d "\n" | grep -o "T" | wc -l)
g=$(grep -v "^>" data/fasta/e_coli.fasta | tr -d "\n" | grep -o "G" | wc -l)
c=$(grep -v "^>" data/fasta/e_coli.fasta | tr -d "\n" | grep -o "C" | wc -l)

# Calculating the (A + T)/ (G + C) ratio
ratio=$(echo "($a + $t) / ($c + $g)" | bc -l)
#note: ratio=$ = store the output of the command as ratio
#note: ($a + $t) / ($c + $g)" = ratio formula
#note: bc -l = calls for calculator

echo "$ratio"
#note: print the output that was stored as ratio
#note:  the ratio is .96922822848944879586
